如果你想更新表 schema，可以在 `schema.ts` 文件中进行。例如，让我们为 `users_table` 添加一个新列 `phone`：

```typescript copy filename="src/db/schema.ts" {16}
// 由内省生成的表 schema
import {
  sqliteTable,
  uniqueIndex,
  integer,
  text,
} from "drizzle-orm/sqlite-core";

export const usersTable = sqliteTable(
  "users_table",
  {
    id: integer().primaryKey({ autoIncrement: true }).notNull(),
    name: text().notNull(),
    age: integer().notNull(),
    email: text().notNull(),
    phone: text(),
  },
  (table) => [
    uniqueIndex("users_table_email_unique").on(table.email)
  ]
);
```
